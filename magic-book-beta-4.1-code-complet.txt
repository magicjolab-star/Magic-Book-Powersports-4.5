name: Build and Deploy Magic Book Beta 4.1

on:
  push:
    branches: [beta-release]
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: magic-book-beta-${{ github.ref }}
  cancel-in-progress: false

jobs:
  android-beta:
    if: github.ref == 'refs/heads/beta-release'
    runs-on: ubuntu-latest
    environment: beta
    timeout-minutes: 40
    env:
      APP_ID: ${{ vars.ANDROID_APP_ID }}
      APP_VERSION_NAME: '4.1.0'
      ANDROID_VERSION_CODE_BASE: ${{ vars.ANDROID_VERSION_CODE_BASE }}
    steps:
      - name: Checkout
        uses: actions/checkout@v7.0.1
        with:
          persist-credentials: false

      - name: Node 24
        uses: actions/setup-node@v7.0.0
        with:
          node-version: '24'
          package-manager-cache: false

      - name: Validate configuration
        shell: bash
        run: |
          set -euo pipefail
          test -n "$APP_ID" || { echo 'Configure ANDROID_APP_ID.'; exit 1; }
          test -f package-lock.json || {
            echo 'Run npm install locally and commit package-lock.json before the first push.'
            exit 1
          }
          node scripts/ci-version.mjs

      - name: Install locked dependencies
        run: npm ci

      - name: Unit tests
        run: npm test

      - name: Inject visual assets, typecheck and build web
        run: npm run build

      - name: Archive web build
        uses: actions/upload-artifact@v7.0.1
        with:
          name: magic-book-beta41-web-${{ github.run_number }}
          path: dist/
          if-no-files-found: error
          retention-days: 14

      - name: Java 21
        uses: actions/setup-java@v5.7.0
        with:
          distribution: temurin
          java-version: '21'

      - name: Android SDK 36
        shell: bash
        run: |
          set -euo pipefail
          SDKMANAGER="$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager"
          test -x "$SDKMANAGER"
          "$SDKMANAGER" --licenses < <(printf 'y\n%.0s' {1..100}) > /dev/null
          "$SDKMANAGER" 'platforms;android-36' 'build-tools;36.0.0' 'platform-tools'

      - name: Prepare Capacitor and native icons
        run: node scripts/prepare-native.mjs android

      - name: Build signed APK and AAB
        shell: bash
        env:
          KEYSTORE_BASE64: ${{ secrets.ANDROID_KEYSTORE_BASE64 }}
          ANDROID_KEYSTORE_PASSWORD: ${{ secrets.ANDROID_KEYSTORE_PASSWORD }}
          ANDROID_KEY_ALIAS: ${{ secrets.ANDROID_KEY_ALIAS }}
          ANDROID_KEY_PASSWORD: ${{ secrets.ANDROID_KEY_PASSWORD }}
        run: |
          set -euo pipefail
          umask 077
          export ANDROID_KEYSTORE_PATH="$RUNNER_TEMP/magic-book-release.jks"
          trap 'rm -f "$ANDROID_KEYSTORE_PATH"' EXIT
          python3 - <<'PYCODE'
          import base64, os
          for name in ('KEYSTORE_BASE64', 'ANDROID_KEYSTORE_PASSWORD', 'ANDROID_KEY_ALIAS', 'ANDROID_KEY_PASSWORD'):
              if not os.environ.get(name):
                  raise SystemExit('Missing secret: ' + name)
          with open(os.environ['ANDROID_KEYSTORE_PATH'], 'wb') as stream:
              stream.write(base64.b64decode(os.environ['KEYSTORE_BASE64'], validate=True))
          PYCODE
          chmod +x android/gradlew
          cd android
          ./gradlew --no-daemon assembleRelease bundleRelease
          cd ..
          mkdir -p release
          cp android/app/build/outputs/apk/release/app-release.apk release/magic-book-beta41.apk
          cp android/app/build/outputs/bundle/release/app-release.aab release/magic-book-beta41.aab
          "$ANDROID_HOME/build-tools/36.0.0/apksigner" verify release/magic-book-beta41.apk

      - name: Archive native packages
        uses: actions/upload-artifact@v7.0.1
        with:
          name: magic-book-beta41-android-${{ github.run_number }}
          path: release/
          if-no-files-found: error
          retention-days: 14

      - name: Distribute APK to Firebase testers (not Google Play or TestFlight)
        shell: bash
        env:
          FIREBASE_APP_ID: ${{ secrets.FIREBASE_APP_ID }}
          FIREBASE_SERVICE_ACCOUNT_JSON: ${{ secrets.FIREBASE_SERVICE_ACCOUNT_JSON }}
          FIREBASE_GROUPS: ${{ vars.FIREBASE_GROUPS || 'beta-testers' }}
        run: |
          set -euo pipefail
          umask 077
          export GOOGLE_APPLICATION_CREDENTIALS="$RUNNER_TEMP/magic-firebase-service-account.json"
          trap 'rm -f "$GOOGLE_APPLICATION_CREDENTIALS"' EXIT
          python3 - <<'PYCODE'
          import json, os
          if not os.environ.get('FIREBASE_APP_ID'):
              raise SystemExit('Missing FIREBASE_APP_ID.')
          data = json.loads(os.environ.get('FIREBASE_SERVICE_ACCOUNT_JSON', '{}'))
          if data.get('type') != 'service_account':
              raise SystemExit('Missing or invalid Firebase service account.')
          with open(os.environ['GOOGLE_APPLICATION_CREDENTIALS'], 'w') as stream:
              json.dump(data, stream)
          PYCODE
          npx --yes --package=firebase-tools@15.30.0 firebase appdistribution:distribute \
            release/magic-book-beta41.apk \
            --app "$FIREBASE_APP_ID" \
            --groups "$FIREBASE_GROUPS" \
            --release-notes "Beta 4.1 - Official visuals, video splash and redesigned interface. Build $ANDROID_VERSION_CODE." \
            --non-interactive

      - name: Final credential cleanup
        if: always()
        shell: bash
        run: rm -f "$RUNNER_TEMP/magic-book-release.jks" "$RUNNER_TEMP/magic-firebase-service-account.json"

# Magic Book Powersports - Beta 4.1

Base d'interface autonome React + TypeScript + Vite + Capacitor, avec les visuels fournis.
Ce projet ne remplace pas automatiquement votre application existante. Le moteur IA,
l'authentification, le backend, les achats et les données de production ne sont pas inclus.
Ne publiez pas cette base à la place de votre application complète avant d'avoir fusionné
et testé ces fonctions existantes. Aucun déploiement n'a été effectué.

## Ressources déjà incluses

| Pièce fournie | Fichier source | Utilisation |
|---|---|---|
| 18490.png / img1 | assets/new/img1.png | Logo de Magic Book, icônes web et natives |
| 18485.png / img2 | assets/reference/img2.png | Référence de la direction artistique |
| 18478.png / img3 | assets/reference/img3.png | Référence marine, or et cyan |
| 18491.mp4 / img4 | assets/new/img4.mp4 | Vidéo du splash, fichier original inchangé |
| 18462.png / img5 | assets/new/img5.png | Logo d'entreprise du footer |

Les deux maquettes de référence ne sont pas chargées dans l'application finale.
Certaines images fournies ont une extension PNG mais des octets JPEG. Le script les
normalise en vrais fichiers PNG pour le navigateur, sans redessiner leur contenu.
Les logos natifs sont redimensionnés depuis img1. Aucun logo de remplacement n'est inventé.

## Premier lancement web

Installer Node 24 (le projet accepte Node >= 22.12), puis, depuis ce dossier :

```sh
npm install
npm test
npm run dev
```

Ouvrir http://localhost:5173. Le script predev prépare automatiquement les ressources.
La première installation crée package-lock.json : le conserver et le committer.
Le registre npm n'était pas accessible dans l'environnement de livraison, donc aucun
package-lock.json fictif n'est fourni. La CI utilise npm ci et exige ce fichier.

```sh
npm run build
npm run preview
```

Le build est produit dans dist/. Ne pas ouvrir directement le index.html source :
il importe du TSX, transformé par Vite. Aucun CDN de polices ou script tiers n'est utilisé.
Le manifeste web est fourni. Aucun service worker ni cache hors ligne web n'est promis.
L'enveloppe Capacitor embarque les fichiers web et la vidéo localement.

## Composants

- src/App.tsx : orchestration démarrage, erreurs, transition et relecture.
- src/theme/theme.ts : couleurs et temporisations centralisées.
- src/theme/global.css : interface responsive, glassmorphism, styles mobile et dialogue.
- src/components/VideoSplash.tsx : autoplay silencieux, repli, arrêt et transition.
- src/screens/HomeScreen.tsx : accueil, navigation, dossiers locaux et paramètres.
- src/components/NewDraftDialog.tsx : saisie d'un brouillon, NIV facultatif et validation.
- src/components/AppFooter.tsx : logo officiel cliquable vers https://magic-app.ca.
- src/lib/bootstrap.ts : point de raccordement au démarrage réel de votre application.
- src/lib/drafts.ts : stockage de prévisualisation isolé, sans moteur IA.

### Splash

Le système affiche d'abord son écran natif statique, avec l'icône officielle.
Dès le premier rendu React, la vidéo prend le relais. Elle ne remplace pas l'écran
du système avant que le moteur de l'application existe.

Le fichier dure environ 9,92 secondes ; l'application n'impose pas de le regarder
entièrement. La durée minimale de l'introduction est de 1,4 seconde. Si le démarrage
est prêt, une transition de 380 ms révèle le tableau de bord. La vidéo est arrêtée
après 7 secondes si le chargement tarde ; le logo sert alors de repli. Une erreur de
lecture ou un autoplay refusé n'empêche pas l'entrée dans l'application.

Après 8,5 secondes de démarrage sans réponse, un écran permet de réessayer ou
d'ouvrir une session temporaire. Il ne fait pas croire que le backend est prêt.
Le mode de réduction des animations utilise directement le logo statique.

### Dossiers de test

Créer et supprimer un brouillon fonctionne. Les compteurs correspondent aux dossiers
réellement saisis, sans données commerciales inventées. Aucun prix n'est calculé.
Le stockage local de prévisualisation n'est pas chiffré : ne pas y mettre de données
personnelles ou sensibles. L'interface indique clairement ce mode.

## Android

Copier .env.example vers .env, puis renseigner APP_ID avec le package EXACT déjà publié,
ANDROID_VERSION_CODE avec un entier plus grand que tous les anciens codes et conserver
APP_VERSION_NAME=4.1.0. L'identifiant n'est volontairement pas inventé.

Prérequis : Node 24, Java 21 et Android SDK 36.

```sh
npm run android:prepare
npm run android:open
```

Le script crée l'enveloppe si elle est absente, synchronise le build, injecte les icônes
et écrit les numéros de version non secrets. La signature release est fournie uniquement
par l'environnement. Le build debug peut être lancé dans Android Studio.

Pour un projet Android existant, intégrer ces composants sur une branche de travail et
préserver ses plugins, manifestes, liens profonds et configurations de signature.
Ne pas lancer une migration majeure de Capacitor sans audit de ces dépendances.
Les dossiers android/ et ios/ sont ignorés dans cette base générée ; pour un projet natif
personnalisé, les versionner dans votre dépôt en adaptant .gitignore.

## iOS

Sur macOS avec Xcode 26 ou plus récent, renseigner APP_ID avec le bundle identifier
réel de l'app iOS et IOS_BUILD_NUMBER avec un nouveau numéro de build.

```sh
npm run ios:prepare
npm run ios:open
```

Le script injecte les icônes et les versions dans le projet Xcode. Choisir votre équipe
Apple et configurer la signature, puis archiver et distribuer via App Store Connect.
Aucun certificat Apple ou profil de provisioning n'est inclus. Aucun workflow TestFlight
prétendument fonctionnel sans ces configurations n'est ajouté.

## Pipeline GitHub Actions

Voir docs/CI-CD.md. Le workflow beta-release.yml produit un build web, un APK Android
signé et un AAB signé. Il distribue l'APK au groupe Firebase beta-testers.
Firebase App Distribution n'est ni une publication Google Play ni un envoi TestFlight.

## Vérifications

Voir docs/VERIFICATION.md pour la distinction entre les tests exécutés et les
compilations / tests natifs restant à exécuter avec vos dépendances et identifiants.

# CI/CD - Beta 4.1

## Déclenchement

Push sur beta-release. Une exécution manuelle est également possible sur cette branche.
Les pull requests vers beta-release ne reçoivent pas de secrets de signature et
exécutent seulement quality.yml (tests et build web).

## Installation initiale

```sh
npm install
npm test
npm run build
```

Committer package-lock.json avant la première CI. Les principales dépendances sont
épinglées dans package.json ; le lockfile fixe aussi leurs dépendances transitives.
Le workflow s'arrête avec un message clair si le lockfile ou le package Android manque.

## Environnement GitHub beta

Créer un environnement beta, limiter les branches autorisées à beta-release et
configurer les valeurs suivantes. Protéger aussi les modifications du workflow.

Variables :

| Nom | Valeur |
|---|---|
| ANDROID_APP_ID | Package Android exact de votre application existante |
| ANDROID_VERSION_CODE_BASE | Entier supérieur à tous les versionCode déjà utilisés |
| FIREBASE_GROUPS | Alias des groupes séparés par une virgule ; défaut beta-testers |

Secrets :

| Nom | Valeur |
|---|---|
| ANDROID_KEYSTORE_BASE64 | Keystore encodé en Base64, sans retour à la ligne |
| ANDROID_KEYSTORE_PASSWORD | Mot de passe du keystore |
| ANDROID_KEY_ALIAS | Alias de la clé |
| ANDROID_KEY_PASSWORD | Mot de passe de la clé |
| FIREBASE_APP_ID | Identifiant Firebase de l'app Android, pas le package name |
| FIREBASE_SERVICE_ACCOUNT_JSON | JSON d'un compte de service autorisé à distribuer |

Le compte de service doit avoir les droits Firebase App Distribution nécessaires
sur le projet concerné (rôle Firebase App Distribution Admin), pas un rôle Owner global.
Enregistrer l'application avec le même package Android dans Firebase et créer le groupe.
Ne pas publier les clés ou comptes de service dans GitHub, public/, src/ ou une variable VITE_*.

Encodage local du keystore, avec Python 3, sur votre propre ordinateur :

```sh
python -c "import base64,pathlib; print(base64.b64encode(pathlib.Path('VOTRE_FICHIER.keystore').read_bytes()).decode())"
```

Copier la sortie uniquement dans le secret GitHub prévu. Ne pas la partager dans un chat.
Le workflow écrit les fichiers sensibles dans RUNNER_TEMP puis les supprime.
Les actions sont fixées à des versions explicites ; un durcissement supplémentaire
consiste à les épingler aux SHA validés par votre organisation.

## Ce que fait le pipeline

Validation des variables, npm ci, tests, injection des ressources, contrôle TypeScript,
build Vite, ajout/synchronisation Capacitor, icônes natives, assembleRelease,
bundleRelease, vérification de signature APK, archivage des trois livrables,
distribution de l'APK via la CLI officielle Firebase.

Le versionCode est ANDROID_VERSION_CODE_BASE + github.run_number. Conserver la base
ou l'augmenter lorsque nécessaire, et ne jamais la réinitialiser vers un ancien numéro.
Une réexécution du même run conserve le même versionCode.

## Google Play et TestFlight

Le fichier AAB est archivé mais ce workflow ne l'envoie pas au Play Store.
Choisir ensuite une piste de test dans la Play Console, ou ajouter un pipeline
Android Publisher API / Fastlane avec les autorisations de ce compte.

Attention : une clé d'upload Google Play peut être différente de la clé qui signe
les APK installés depuis le Play Store. Un APK Firebase signé avec la seule clé
d'upload peut ne pas pouvoir remplacer une app installée depuis Google Play.
Utiliser un appareil/profil de test distinct dans ce cas. Ne pas désinstaller une
app avec des données importantes sans sauvegarde.

Un envoi TestFlight nécessite une compilation iOS sur macOS/Xcode, une signature
Apple et un upload vers App Store Connect. Le job Linux/Firebase n'effectue pas
ces opérations. ios:prepare prépare les ressources et la version, pas la distribution.

import { useEffect, useRef, useState } from 'react';
import { appConfig } from '../config';
import { theme } from '../theme/theme';
import { canExitSplash } from '../lib/policy';

interface Props { ready: boolean; onComplete: () => void }

export function VideoSplash({ ready, onComplete }: Props) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const [minimumElapsed, setMinimumElapsed] = useState(false);
  const [skip, setSkip] = useState(false);
  const [fallback, setFallback] = useState(() => window.matchMedia('(prefers-reduced-motion: reduce)').matches);
  const [leaving, setLeaving] = useState(false);

  useEffect(() => {
    const minimum = window.setTimeout(() => setMinimumElapsed(true), theme.splash.minimumMs);
    const maximum = window.setTimeout(() => setFallback(true), theme.splash.maximumVideoMs);
    return () => { window.clearTimeout(minimum); window.clearTimeout(maximum); };
  }, []);

  useEffect(() => {
    const video = videoRef.current;
    if (!video || fallback) return;
    let active = true;
    video.muted = true;
    video.playsInline = true;
    video.play().catch(() => { if (active) setFallback(true); });
    return () => { active = false; video.pause(); };
  }, [fallback]);

  useEffect(() => {
    if (canExitSplash(ready, minimumElapsed, skip)) setLeaving(true);
  }, [ready, minimumElapsed, skip]);

  useEffect(() => {
    if (!leaving) return;
    // Timer is a fallback independent of transitionend, which may not fire.
    const timer = window.setTimeout(onComplete, theme.splash.fadeMs);
    return () => window.clearTimeout(timer);
  }, [leaving, onComplete]);

  return (
    <section className={`splash ${leaving ? 'splash--leaving' : ''}`} aria-label="Chargement de Magic Book">
      <div className="splash__frame">
        {fallback ? (
          <img className="splash__fallback" src={appConfig.assets.appIcon} alt="Magic Book Powersports" />
        ) : (

import { useEffect, useState } from 'react';
import { bootstrapApp } from '../lib/bootstrap';
import type { VehicleDraft } from '../lib/drafts';
import { theme } from '../theme/theme';

export type BootState =
  | { phase: 'loading' }
  | { phase: 'ready'; drafts: VehicleDraft[]; memoryOnly: boolean }
  | { phase: 'error' };

export function useBootstrap(attempt: number) {
  const [state, setState] = useState<BootState>({ phase: 'loading' });
  useEffect(() => {
    let active = true;
    const controller = new AbortController();
    setState({ phase: 'loading' });
    const timer = window.setTimeout(() => {
      if (active) {
        controller.abort();
        setState({ phase: 'error' });
      }
    }, theme.bootstrapTimeoutMs);
    bootstrapApp(controller.signal).then(drafts => {
      if (active && !controller.signal.aborted) setState({ phase: 'ready', drafts, memoryOnly: false });
    }).catch(() => {
      if (active && !controller.signal.aborted) setState({ phase: 'error' });
    }).finally(() => window.clearTimeout(timer));
    return () => { active = false; controller.abort(); window.clearTimeout(timer); };
  }, [attempt]);
  return { state, continueInMemory: () => setState({ phase: 'ready', drafts: [], memoryOnly: true }) };
}

export interface VehicleDraft {
  id: string;
  category: string;
  brand: string;
  model: string;
  year: number;
  vin: string;
  createdAt: string;
}
export const STORAGE_KEY = 'magic-book.beta41.vehicle-drafts.v1';
const MAX_DRAFTS = 250;

export function parseDrafts(raw: string | null): VehicleDraft[] {
  if (raw === null) return [];
  const data: unknown = JSON.parse(raw);
  if (!Array.isArray(data) || data.length > MAX_DRAFTS) throw new Error('Dossiers locaux invalides.');
  return data.map((value: unknown) => {
    if (!value || typeof value !== 'object') throw new Error('Dossier invalide.');
    const d = value as Record<string, unknown>;
    for (const key of ['id', 'category', 'brand', 'model', 'vin', 'createdAt']) {
      if (typeof d[key] !== 'string' || (d[key] as string).length > 160) throw new Error('Champ invalide.');
    }
    if (!Number.isInteger(d.year) || Number(d.year) < 1900 || Number(d.year) > 2100 ||
        !Number.isFinite(Date.parse(d.createdAt as string))) throw new Error('Date invalide.');
    return { id: d.id as string, category: d.category as string, brand: d.brand as string,
      model: d.model as string, year: d.year as number, vin: d.vin as string, createdAt: d.createdAt as string };
  });
}

export function loadDrafts(): VehicleDraft[] {
  return parseDrafts(window.localStorage.getItem(STORAGE_KEY));
}

export function saveDrafts(drafts: VehicleDraft[]): void {
  if (drafts.length > MAX_DRAFTS) throw new Error('Limite de 250 dossiers atteinte.');
  window.localStorage.setItem(STORAGE_KEY, JSON.stringify(drafts));
}

export function createDraft(values: Omit<VehicleDraft, 'id' | 'createdAt'>): VehicleDraft {
  return { ...values, id: crypto.randomUUID(), createdAt: new Date().toISOString() };
}

export function releaseVersion(base: string, run: string): number {
  if (!/^\d+$/.test(base) || !/^\d+$/.test(run)) throw new Error('Version invalide.');
  const value = Number(base) + Number(run);
  if (!Number.isSafeInteger(value) || value < 1 || value > 2100000000) {
    throw new Error('versionCode hors limites.');
  }
  return value;
}

export function validateVin(value: string): boolean {

}

export const theme = {
  colors: {
    navy: '#030b17', surface: '#0b192b', surfaceRaised: '#10233b',
    gold: '#d3ae65', goldLight: '#f4dca0', cyan: '#62dcff',
    white: '#f6f8fc', muted: '#b9c7d9', danger: '#ffb4b4',
  },
  splash: { minimumMs: 1400, maximumVideoMs: 7000, fadeMs: 380 },
  bootstrapTimeoutMs: 8500,
} as const;

export function applyTheme(): void {
  for (const [key, value] of Object.entries(theme.colors)) {
    document.documentElement.style.setProperty(`--${key}`, value);
  }
  document.documentElement.style.setProperty('--splash-fade', `${theme.splash.fadeMs}ms`);
}

import test from 'node:test';
import assert from 'node:assert/strict';
import { canExitSplash, releaseVersion, validateVin } from '../src/lib/policy.ts';
import { parseDrafts } from '../src/lib/drafts.ts';

test('splash never enters a dashboard that is not ready',()=>{
  assert.equal(canExitSplash(false,true,true),false);
});
test('splash waits for minimum exposure',()=>{
  assert.equal(canExitSplash(true,false,false),false);
  assert.equal(canExitSplash(true,true,false),true);
});
test('skip bypasses the intro but not application readiness',()=>{
  assert.equal(canExitSplash(true,false,true),true);
  assert.equal(canExitSplash(false,false,true),false);
});
test('version codes increase with runs',()=>{
  assert.equal(releaseVersion('50000','12'),50012);
  assert.equal(releaseVersion('50000','13'),50013);
});
test('invalid and overflowing version codes are rejected',()=>{
  for(const value of ['','4.1','-1','not-a-number']) assert.throws(()=>releaseVersion(value,'1'));
  assert.throws(()=>releaseVersion('2100000000','1'));
});
test('optional VIN is validated without forbidden letters',()=>{
  assert.equal(validateVin(''),true);
  assert.equal(validateVin('1HGCM82633A004352'),true);
  assert.equal(validateVin('1HGCM82633A00I352'),false);
  assert.equal(validateVin('123'),false);
});
test('empty local storage produces zero real dossiers, not demo results',()=>{
  assert.deepEqual(parseDrafts(null),[]);
});
test('invalid stored JSON is surfaced, not silently overwritten',()=>{
  assert.throws(()=>parseDrafts('{broken'));
  assert.throws(()=>parseDrafts('{"items":[]}'));
});
test('draft fields are sanitized by reconstruction',()=>{
  const raw=[{id:'d1',category:'VTT',brand:'Yamaha',model:'Grizzly',year:2026,vin:'',createdAt:'2026-09-12T12:00:00.000Z',unknown:'removed'}];
  const result=parseDrafts(JSON.stringify(raw));
  assert.equal(result[0].brand,'Yamaha');
  assert.equal('unknown' in result[0],false);
});
test('oversized draft collections and invalid years are rejected',()=>{
  assert.throws(()=>parseDrafts(JSON.stringify(new Array(251).fill({}))));
  assert.throws(()=>parseDrafts(JSON.stringify([{id:'d1',category:'VTT',brand:'Yamaha',model:'X',year:0,vin:'',createdAt:'invalid'}])));
});
